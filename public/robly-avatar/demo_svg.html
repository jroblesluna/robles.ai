Vectorial, continua (sin steps) y con transicion suave entre estados — cada pose es un SVG independiente y autonomo.

Estado actual: idle

Integracion

<div id="robly-stage">
  <!-- pega el contenido de robly-idle.svg, robly-listening.svg, etc. -->
  <!-- o cargalos con fetch()/<object> e intercambia cual esta visible -->
</div>

<script>
  function showState(name) {
    document.querySelectorAll('#robly-stage > *').forEach(el =>
      el.classList.toggle('active', el.dataset.state === name));
  }
  showState('listening');  // el usuario esta escribiendo
  showState('thinking');   // esperando la respuesta
  showState('speaking');   // la respuesta se esta escribiendo
  showState('idle');       // en reposo
</script>
